cmake_minimum_required(VERSION 3.16)
project(g1_controller LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(DEPLOY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(ZLIB REQUIRED)
find_package(Threads REQUIRED)
find_package(Boost REQUIRED COMPONENTS program_options)
find_package(fmt REQUIRED)
find_package(OpenSSL REQUIRED)
find_package(jsoncpp CONFIG REQUIRED)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory(${DEPLOY_ROOT}/vendor/unitree_sdk2 ${CMAKE_BINARY_DIR}/sdk)
if(CMAKE_SYSTEM_PROCESSOR STREQUAL "x86_64")
  set(ORT_ARCH x64)
elseif(CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
  set(ORT_ARCH aarch64)
else()
  message(FATAL_ERROR "Unsupported architecture: ${CMAKE_SYSTEM_PROCESSOR}")
endif()
set(ORT_DIR ${DEPLOY_ROOT}/deploy/thirdparty/onnxruntime-linux-${ORT_ARCH}-1.22.0)
add_library(onnxruntime SHARED IMPORTED)
set_target_properties(onnxruntime PROPERTIES IMPORTED_LOCATION ${ORT_DIR}/lib/libonnxruntime.so.1.22.0 INTERFACE_INCLUDE_DIRECTORIES ${ORT_DIR}/include)
set(MUJOCO_DIR ${DEPLOY_ROOT}/simulate/mujoco CACHE PATH "MuJoCo C distribution")
find_library(MUJOCO_LIBRARY NAMES mujoco libmujoco.so.3.3.6 PATHS ${MUJOCO_DIR}/lib NO_DEFAULT_PATH REQUIRED)
add_library(mujoco SHARED IMPORTED)
set_target_properties(mujoco PROPERTIES IMPORTED_LOCATION ${MUJOCO_LIBRARY} INTERFACE_INCLUDE_DIRECTORIES ${MUJOCO_DIR}/include)
add_library(cnpy_lib STATIC ${DEPLOY_ROOT}/deploy/thirdparty/cnpy/cnpy.cpp)
target_include_directories(cnpy_lib PUBLIC ${DEPLOY_ROOT}/deploy/thirdparty/cnpy)
target_link_libraries(cnpy_lib PUBLIC ZLIB::ZLIB)
add_library(g1_policy_runtime STATIC src/control/MotionPolicy.cpp src/control/LegacyPolicy.cpp src/control/Registry.cpp)
target_include_directories(g1_policy_runtime PUBLIC include)
target_link_libraries(g1_policy_runtime PUBLIC Eigen3::Eigen yaml-cpp onnxruntime cnpy_lib OpenSSL::Crypto)
target_compile_options(g1_policy_runtime PRIVATE -Wall -Wextra -ffp-contract=off)
add_library(g1_control STATIC src/control/Controller.cpp src/control/Physics.cpp src/control/Session.cpp src/control/RunLog.cpp)
target_link_libraries(g1_control PUBLIC g1_policy_runtime mujoco jsoncpp_lib)
target_compile_options(g1_control PRIVATE -Wall -Wextra)
find_package(OpenGL REQUIRED COMPONENTS OpenGL EGL GLX)
add_executable(g1_deploy cpp_main.cpp src/control/Robot.cpp src/control/VideoRecorder.cpp)
target_link_libraries(g1_deploy PRIVATE g1_control unitree_sdk2 Boost::program_options OpenGL::EGL OpenGL::GLX OpenGL::OpenGL)
target_compile_definitions(g1_deploy PRIVATE G1_DEPLOY_ROOT="${DEPLOY_ROOT}")
find_package(glfw3 QUIET)
find_package(OpenGL QUIET)
if(TARGET glfw AND OpenGL_FOUND)
  target_sources(g1_deploy PRIVATE src/control/Viewer.cpp)
  target_link_libraries(g1_deploy PRIVATE glfw OpenGL::GL)
  target_compile_definitions(g1_deploy PRIVATE G1_VIEWER=1)
endif()
add_executable(g1_prepare prepare_main.cpp)
target_link_libraries(g1_prepare PRIVATE unitree_sdk2 yaml-cpp Boost::program_options)
# Keep the original C++ controller and every original action available.
add_executable(g1_ctrl main.cpp src/State_RLBase.cpp src/State_Mimic.cpp)
target_include_directories(g1_ctrl PRIVATE include ${DEPLOY_ROOT}/deploy/include)
target_link_libraries(g1_ctrl PRIVATE g1_policy_runtime unitree_sdk2 Boost::program_options fmt::fmt)
set_target_properties(g1_ctrl PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/build)
include(CTest)
if(BUILD_TESTING)
  foreach(probe legacy motion)
    add_executable(${probe}_probe ${DEPLOY_ROOT}/tests/cpp_migration/${probe}_probe.cpp)
    target_link_libraries(${probe}_probe PRIVATE g1_policy_runtime)
  endforeach()
  add_executable(motion_npz_probe ${DEPLOY_ROOT}/tests/cpp_migration/motion_npz_probe.cpp)
  target_link_libraries(motion_npz_probe PRIVATE cnpy_lib)
  add_executable(prepare_checks ${DEPLOY_ROOT}/tests/cpp_migration/prepare_checks.cpp)
  target_link_libraries(prepare_checks PRIVATE unitree_sdk2 yaml-cpp Boost::program_options)
  target_compile_options(prepare_checks PRIVATE -Wno-terminate)
  add_test(NAME prepare_offline COMMAND prepare_checks)
  add_executable(video_probe ${DEPLOY_ROOT}/tests/cpp_migration/video_probe.cpp src/control/VideoRecorder.cpp)
  target_link_libraries(video_probe PRIVATE g1_control OpenGL::EGL OpenGL::GLX OpenGL::OpenGL Threads::Threads)
  find_program(FFMPEG_EXECUTABLE ffmpeg)
  if(FFMPEG_EXECUTABLE)
    add_test(NAME offscreen_video COMMAND video_probe ${DEPLOY_ROOT} ${CMAKE_BINARY_DIR}/video_test.mp4)
  endif()
  if(TARGET glfw AND OpenGL_FOUND)
    add_executable(viewer_checks ${DEPLOY_ROOT}/tests/cpp_migration/viewer_checks.cpp)
    target_link_libraries(viewer_checks PRIVATE g1_control glfw OpenGL::GL)
    add_test(NAME viewer_interaction COMMAND viewer_checks ${DEPLOY_ROOT})
    set_tests_properties(viewer_interaction PROPERTIES SKIP_RETURN_CODE 77)
  endif()
  add_executable(loopback_g1 ${DEPLOY_ROOT}/tests/cpp_migration/loopback_g1.cpp)
  target_link_libraries(loopback_g1 PRIVATE unitree_sdk2 Threads::Threads)
  add_executable(session_probe ${DEPLOY_ROOT}/tests/cpp_migration/session_probe.cpp)
  target_link_libraries(session_probe PRIVATE g1_control)
  add_executable(registry_probe ${DEPLOY_ROOT}/tests/cpp_migration/registry_probe.cpp)
  target_link_libraries(registry_probe PRIVATE g1_policy_runtime)
  add_executable(robot_checks ${DEPLOY_ROOT}/tests/cpp_migration/robot_checks.cpp)
  target_link_libraries(robot_checks PRIVATE g1_control unitree_sdk2)
  target_compile_options(robot_checks PRIVATE -Wno-subobject-linkage -Wno-terminate)
  add_test(NAME robot_packets_offline COMMAND robot_checks)
  add_executable(controller_test tests/controller_test.cpp)
  target_link_libraries(controller_test PRIVATE g1_control)
  add_test(NAME hierarchy COMMAND controller_test)
  find_package(Python3 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME full_session_integration COMMAND ${Python3_EXECUTABLE}
      ${DEPLOY_ROOT}/tests/cpp_migration/session_integration.py
      --binary $<TARGET_FILE:g1_deploy> --root ${DEPLOY_ROOT})
    add_test(NAME loopback_integration COMMAND ${Python3_EXECUTABLE}
      ${DEPLOY_ROOT}/tests/cpp_migration/loopback_integration.py
      --binary $<TARGET_FILE:g1_deploy> --root ${DEPLOY_ROOT} --fixture $<TARGET_FILE:loopback_g1>)
    set_tests_properties(full_session_integration PROPERTIES TIMEOUT 240)
    set_tests_properties(loopback_integration PROPERTIES RUN_SERIAL TRUE TIMEOUT 120)
    add_test(NAME interactive_sim2sim COMMAND ${Python3_EXECUTABLE}
      ${DEPLOY_ROOT}/tests/cpp_migration/interactive_simulation.py
      --binary $<TARGET_FILE:g1_deploy> --root ${DEPLOY_ROOT})
  endif()
  add_test(NAME actors_load_offline COMMAND g1_deploy --root ${DEPLOY_ROOT} --inspect)
  foreach(scenario switching extensions)
    add_test(NAME simulation_${scenario} COMMAND ${CMAKE_COMMAND}
      -DBINARY=$<TARGET_FILE:g1_deploy> -DROOT=${DEPLOY_ROOT} -DSCENARIO=${scenario}
      -P ${DEPLOY_ROOT}/tests/cpp_migration/check_simulation.cmake)
  endforeach()
endif()
